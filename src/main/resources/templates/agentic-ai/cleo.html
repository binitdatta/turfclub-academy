Escalation levels

Friendly notice for overdue balances. The implementation can send reminder letters automatically subject to configuration.

Higher-impact correspondence that requires manager approval in the declared employee policy.

Changes a customer party status via a restricted ERP endpoint; never grant this action to a generic chatbot token.

Collections workflow, thresholds and actions

"""AI Collections Officer - chases overdue receivables every weekday morning.

Work: reads AR's ageing (every open invoice with its days overdue), groups it by customer and picks the customers due
a letter: nobody contacted in the last COLLECTIONS_RESEND_DAYS days, most overdue first, at most MAX_CUSTOMERS a run.
The dunning stage is fixed by policy from the oldest invoice: 1-30 days REMINDER, 31-60 SECOND_NOTICE, 61+ FINAL_NOTICE.
The model reads each customer's history (payments, membership, prize money owed to them, earlier letters) and writes
the personal part of each letter - the statement of invoices and amounts is always rendered by code from the ERP.
Policy:
  * REMINDER and SECOND_NOTICE -> AUTO: sent at once (unless the text quotes an amount that is not on the statement);
  * FINAL_NOTICE -> APPROVAL by an AR manager (AR_MANAGE), who can edit the letter;
  * account hold (party BLOCKED) when 90+ days and at least $500 overdue, or still 61+ days overdue 14 days after a final
    notice -> always APPROVAL by an AR manager.
Write: POST /api/v1/parties/{id}/status as service-account-ai-collections (role SVC_ACCOUNT_HOLD) - nothing else. The
letters are recorded in the workforce schema and sent through the mailer (logged by default).
"""
from __future__ import annotations

import re
from decimal import ROUND_HALF_UP, Decimal
from pathlib import Path
from typing import Any

from ...db import from_json
from ...platform.api import ApiError
from ...platform.mailer import MailError
from ..base import (APPROVAL, AUTO, SKIP, ActionResult, EditError, EditField, Employee, ExecContext, Facts,
                    PlannedAction)
from .tools import TOOLS

CODE = "collections"
STAGES = ("REMINDER", "SECOND_NOTICE", "FINAL_NOTICE")
STAGE_LABEL = {"REMINDER": "Payment reminder", "SECOND_NOTICE": "Second notice", "FINAL_NOTICE": "Final notice"}


def _d(value: Any) -> Decimal:
    try:
        return Decimal(str(value if value not in (None, "") else 0))
    except Exception:  # noqa: BLE001
        return Decimal(0)


def _money(value: Decimal | float) -> float:
    return float(Decimal(str(value)).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP))


def fmt(value: Any) -> str:
    return f"${_money(_d(value)):,.2f}"


def resend_days(cfg) -> int:
    return cfg.employee_int(CODE, "RESEND_DAYS", 14)


def max_customers(cfg) -> int:
    return cfg.employee_int(CODE, "MAX_CUSTOMERS", 8)


def hold_days(cfg) -> int:
    return cfg.employee_int(CODE, "HOLD_DAYS", 90)


def hold_min(cfg) -> Decimal:
    return cfg.employee_dec(CODE, "HOLD_MIN_AMOUNT", "500")


def auto_send(cfg) -> bool:
    return cfg.employee_bool(CODE, "AUTO_SEND", True)


def stage_for(days: int) -> str:
    return "REMINDER" if days <= 30 else "SECOND_NOTICE" if days <= 60 else "FINAL_NOTICE"


# ----------------------------------------------------------------------------- prepare (deterministic, no LLM)

def history(ctx) -> tuple[dict[int, dict], set[int], set[int]]:
    """Letters per party (last letter, last final notice), parties already put on hold, parties waiting in the inbox."""
    letters = {r["party_id"]: r for r in ctx.db.query(
        "SELECT party_id, MAX(created_at) AS last_at, "
        "MAX(CASE WHEN created_at >= UTC_TIMESTAMP() - INTERVAL %s DAY THEN 1 ELSE 0 END) AS recent, "
        "MAX(CASE WHEN kind = 'FINAL_NOTICE' THEN created_at END) AS last_final "
        "FROM document WHERE employee_code = %s AND party_id IS NOT NULL GROUP BY party_id", (resend_days(ctx.cfg), CODE))}
    held = {int(r["pid"]) for r in ctx.db.query(
        "SELECT JSON_UNQUOTE(JSON_EXTRACT(request_json, '$.partyId')) AS pid FROM action "
        "WHERE employee_code = %s AND kind = 'PLACE_HOLD' AND status = 'DONE'", (CODE,)) if r["pid"]}
    pending = set()
    for r in ctx.db.query("SELECT payload FROM approval WHERE employee_code = %s AND status = 'PENDING'", (CODE,)):
        pid = (from_json(r["payload"]) or {}).get("partyId")
        if pid:
            pending.add(int(pid))
    return letters, held, pending


def prepare(ctx) -> Facts:
    rows = ctx.get("AR", "/ageing/rows", asOf=ctx.today.isoformat(), minDaysOverdue=1) or []
    if not rows:
        return Facts(True, "No receivables invoice is overdue.", {"overdueCustomers": 0})
    customers: dict[int, dict] = {}
    for r in rows:
        c = customers.setdefault(int(r["customerPartyId"]), {"partyId": int(r["customerPartyId"]), "customer": r.get("customer"),
                                                             "totalOverdue": Decimal(0), "maxDaysOverdue": 0, "invoices": []})
        c["totalOverdue"] += _d(r.get("balance"))
        c["maxDaysOverdue"] = max(c["maxDaysOverdue"], int(r.get("daysOverdue") or 0))
        c["invoices"].append({"invoiceId": r["invoiceId"], "invoiceNumber": r["invoiceNumber"], "sourceModule": r.get("sourceModule"),
                              "sourceReference": r.get("sourceReference"), "invoiceDate": r.get("invoiceDate"),
                              "dueDate": r.get("dueDate"), "balance": _money(_d(r.get("balance"))),
                              "daysOverdue": int(r.get("daysOverdue") or 0)})
    letters, held, pending = history(ctx)
    total_all = sum((c["totalOverdue"] for c in customers.values()), Decimal(0))
    due, contacted, waiting = [], 0, 0
    for c in customers.values():
        if c["partyId"] in pending:
            waiting += 1
            continue
        h = letters.get(c["partyId"])
        if h and h.get("recent"):
            contacted += 1
            continue
        c["stage"] = stage_for(c["maxDaysOverdue"])
        c["lastLetterAt"] = h["last_at"].isoformat(timespec="minutes") if h and h.get("last_at") else None
        last_final = h.get("last_final") if h else None
        days_since_final = (ctx.today - last_final.date()).days if last_final else None
        reasons = []
        if c["maxDaysOverdue"] >= hold_days(ctx.cfg) and c["totalOverdue"] >= hold_min(ctx.cfg):
            reasons.append(f"{c['maxDaysOverdue']} days overdue with {fmt(c['totalOverdue'])} outstanding "
                           f"(policy: {hold_days(ctx.cfg)}+ days and at least {fmt(hold_min(ctx.cfg))})")
        if days_since_final is not None and days_since_final >= resend_days(ctx.cfg) and c["maxDaysOverdue"] > 60:
            reasons.append(f"still {c['maxDaysOverdue']} days overdue {days_since_final} days after the final notice")
        c["holdCandidate"] = bool(reasons) and c["partyId"] not in held
        c["holdReason"] = "; ".join(reasons) if c["holdCandidate"] else None
        due.append(c)
    if not due:
        return Facts(True, f"{len(customers)} customer(s) are overdue, but all were contacted in the last "
                           f"{resend_days(ctx.cfg)} days or are waiting for an approval.",
                     {"overdueCustomers": len(customers), "recentlyContacted": contacted, "waitingForApproval": waiting})
    due.sort(key=lambda c: (STAGES.index(c["stage"]), c["totalOverdue"]), reverse=True)
    selected = due[:max_customers(ctx.cfg)]
    for c in selected:
        try:
            party = ctx.get("CORE", f"/parties/{c['partyId']}") or {}
        except ApiError:
            party = {}
        c["email"] = party.get("email")
        c["partyType"] = party.get("type")
        c["partyStatus"] = party.get("status")
        if party.get("status") == "BLOCKED":
            c["holdCandidate"] = False
            c["holdReason"] = None
        c["totalOverdue"] = _money(c["totalOverdue"])
        c["invoices"].sort(key=lambda i: i["daysOverdue"], reverse=True)
    holds = sum(1 for c in selected if c["holdCandidate"])
    value = sum(Decimal(str(c["totalOverdue"])) for c in selected)
    headline = (f"{len(selected)} of {len(customers)} overdue customers are due a letter ({fmt(value)} overdue)"
                + (f"; {holds} eligible for an account hold" if holds else "") + ".")
    return Facts(False, headline, {"asOf": ctx.today.isoformat(), "overdueCustomers": len(customers),
                                   "totalOverdueAllCustomers": _money(total_all), "recentlyContacted": contacted,
                                   "waitingForApproval": waiting, "dueButNotThisRun": max(len(due) - len(selected), 0),
                                   "customers": selected})

CLOSING = {
    "REMINDER": "If you have already paid, thank you - please ignore this reminder. You can pay at the club office or by "
                "bank transfer, quoting the invoice number.",
    "SECOND_NOTICE": "Please arrange payment within 14 days of this letter, or contact Accounts Receivable if there is a "
                     "problem with any of these invoices so we can resolve it together.",
    "FINAL_NOTICE": "This is our final notice. If the balance is not settled within 7 days of this letter, your account "
                    "may be placed on hold, which suspends further credit, bookings and member charges until it is paid. "
                    "Please contact Accounts Receivable straight away if you wish to agree a payment plan.",
}

FALLBACK_OPENING = {
    "REMINDER": "Our records show that the invoices below are now past their due date. This is a friendly reminder in "
                "case they have been overlooked.",
    "SECOND_NOTICE": "We wrote to you recently about the invoices below, which remain unpaid and are now more than 30 days "
                     "past their due date.",
    "FINAL_NOTICE": "Despite our earlier reminders, the invoices below remain unpaid and are now more than 60 days past "
                    "their due date.",
}


def statement(customer: dict, as_of: str) -> str:
    lines = [f"Overdue invoices as of {as_of}:", "",
             f"{'Invoice':<18}{'Due':<12}{'Days':>6}{'Balance':>14}"]
    for i in customer["invoices"]:
        lines.append(f"{i['invoiceNumber']:<18}{str(i['dueDate'])[:10]:<12}{i['daysOverdue']:>6}{fmt(i['balance']):>14}")
    lines.append(f"{'Total overdue':<36}{fmt(customer['totalOverdue']):>14}")
    return "\n".join(lines)


def render_letter(customer: dict, stage: str, opening: str, as_of: str) -> str:
    return (f"Dear {customer['customer']},\n\n{opening.strip()}\n\n{statement(customer, as_of)}\n\n{CLOSING[stage]}\n\n"
            f"Kind regards,\nAccounts Receivable\nRollingstone Turf Club")


AMOUNT = re.compile(r"\$\s?(\d{1,3}(?:,\d{3})+(?:\.\d{1,2})?|\d+(?:\.\d{1,2})?)")


def foreign_amounts(text: str, customer: dict) -> list[str]:
    """Dollar amounts in the model's text that are not on the customer's statement."""
    allowed = {_money(customer["totalOverdue"])} | {_money(i["balance"]) for i in customer["invoices"]}
    out = []
    for m in AMOUNT.finditer(text or ""):
        value = _money(Decimal(m.group(1).replace(",", "")))
        if value not in allowed:
            out.append(m.group(0))
    return out

PROPOSAL_SCHEMA = {
    "type": "object",
    "properties": {
        "summary": {"type": "string", "description": "2-4 sentences for the AR manager: who you are writing to and anything "
                                                     "unusual you found. Plain English, no markdown."},
        "letters": {"type": "array", "description": "One entry per customer on the worklist.",
                    "items": {"type": "object", "properties": {
                        "party_id": {"type": "integer"},
                        "subject": {"type": "string", "description": "E-mail subject line"},
                        "opening": {"type": "string", "description": "The personal paragraph(s) of the letter: 1-3 short "
                                    "paragraphs, no greeting, no sign-off, no invoice table (the statement is added by the "
                                    "system). Quote amounts only exactly as on the worklist."},
                        "skip": {"type": "boolean", "description": "true = do not write to this customer now"},
                        "skip_reason": {"type": "string"},
                        "recommend_hold": {"type": "boolean", "description": "Your view on an account hold for this customer"},
                        "hold_reason": {"type": "string", "description": "Why (or why not) - shown to the AR manager"}},
                        "required": ["party_id"]}},
    },
    "required": ["summary", "letters"],
}


def plan(ctx, facts: Facts, proposal: dict | None) -> list[PlannedAction]:
    cfg = ctx.cfg
    as_of = facts.data.get("asOf") or ctx.today.isoformat()
    by_party = {}
    notes = []
    for entry in (proposal or {}).get("letters") or []:
        try:
            by_party[int(entry.get("party_id"))] = entry
        except (TypeError, ValueError):
            notes.append(f"Ignored a letter with an invalid party id ({entry.get('party_id')!r}).")
    actions: list[PlannedAction] = []
    for c in facts.data.get("customers") or []:
        pid = c["partyId"]
        stage = c["stage"]
        entry = by_party.pop(pid, None)
        reasons: list[str] = []
        if entry is None:
            reasons.append("The AI wrote no letter for this customer, so the standard wording is used." if proposal is not None
                           else "The AI did not produce a proposal, so the standard wording is used.")
            entry = {}
        if entry.get("skip"):
            actions.append(PlannedAction(f"letter-{pid}", stage, f"{STAGE_LABEL[stage]} to {c['customer']} - skipped",
                                         (entry.get("skip_reason") or "").strip() or "The AI decided not to write now.",
                                         {"partyId": pid, "customer": c["customer"], "stage": stage}, SKIP,
                                         [f"Skipped by the AI: {(entry.get('skip_reason') or 'no reason given').strip()}"],
                                         c["totalOverdue"]))
        else:
            opening = (entry.get("opening") or "").strip()[:2500] or FALLBACK_OPENING[stage]
            foreign = foreign_amounts(opening, c)
            needs_person = []
            if stage == "FINAL_NOTICE":
                needs_person.append("Final notices always need an AR manager's approval.")
            elif not auto_send(cfg):
                needs_person.append("Automatic sending is switched off (COLLECTIONS_AUTO_SEND=false).")
            if foreign:
                needs_person.append(f"The letter text quotes {', '.join(foreign)}, which is not on the customer's statement.")
            if not c.get("email") and cfg.MAIL_MODE == "smtp":
                needs_person.append("The customer has no e-mail address in the party register.")
            reasons = needs_person + reasons
            subject = (entry.get("subject") or "").strip()[:200] or f"{STAGE_LABEL[stage]}: overdue account - Rollingstone Turf Club"
            body = render_letter(c, stage, opening, as_of)
            mode = APPROVAL if needs_person else AUTO
            actions.append(PlannedAction(
                f"letter-{pid}", stage, f"{STAGE_LABEL[stage]} to {c['customer']} ({fmt(c['totalOverdue'])})",
                f"{c['maxDaysOverdue']} days overdue on {len(c['invoices'])} invoice(s).",
                {"partyId": pid, "customer": c["customer"], "recipient": c.get("email"), "stage": stage, "subject": subject,
                 "body": body, "totalOverdue": c["totalOverdue"], "maxDaysOverdue": c["maxDaysOverdue"],
                 "invoices": [i["invoiceNumber"] for i in c["invoices"]]},
                mode, reasons, c["totalOverdue"]))
        recommend = entry.get("recommend_hold")
        ai_view = (entry.get("hold_reason") or "").strip()
        if c.get("holdCandidate") or (recommend and c["maxDaysOverdue"] > 60 and c.get("partyStatus") != "BLOCKED"):
            why = c.get("holdReason") or f"Recommended by the AI: {ai_view or 'no reason given'}"
            hold_reasons = [f"Account holds always need an AR manager's approval. Policy: {why}."]
            if ai_view:
                hold_reasons.append(f"The AI's view: {ai_view}" + ("" if recommend is not False else " (it recommends against)"))
            actions.append(PlannedAction(
                f"hold-{pid}", "PLACE_HOLD", f"Account hold on {c['customer']} ({fmt(c['totalOverdue'])} overdue)",
                f"Set the party to BLOCKED in the party register: {why}.",
                {"partyId": pid, "customer": c["customer"], "reason": f"Collections: {why}"[:480],
                 "totalOverdue": c["totalOverdue"], "maxDaysOverdue": c["maxDaysOverdue"]},
                APPROVAL, hold_reasons, c["totalOverdue"]))
        elif recommend:
            notes.append(f"{c['customer']}: the AI recommended a hold, but the account is not eligible "
                         f"({c['maxDaysOverdue']} days overdue).")
    for pid in by_party:
        notes.append(f"Ignored a letter for party {pid}: not on this run's worklist.")
    if notes and actions:
        actions[0].reasons.extend(notes)
    return actions


# ----------------------------------------------------------------------------- execute

def execute(ectx: ExecContext, action: PlannedAction) -> ActionResult:
    p = action.payload
    if action.kind == "PLACE_HOLD":
        try:
            party = ectx.api.post("CORE", f"/parties/{p['partyId']}/status", {"status": "BLOCKED", "reason": p["reason"]})
        except ApiError as e:
            return ActionResult("FAILED", error=str(e), response=e.body)
        return ActionResult("DONE", erp_reference=f"Party {p['partyId']} {party.get('status')}", response=party)
    body = p["body"]
    footer = "\n\n--\nThis letter was prepared by the club's AI Collections Officer"
    footer += f" and approved by {ectx.approved_by}." if ectx.approved_by else "."
    body += footer
    try:
        channel = ectx.mailer.send(p.get("recipient"), p["subject"], body)
    except MailError as e:
        return ActionResult("FAILED", error=str(e))
    return ActionResult("DONE", erp_reference=f"{STAGE_LABEL.get(action.kind, 'Letter')} to {p.get('recipient') or p['customer']}",
                        response={"channel": channel},
                        document={"kind": action.kind, "party_id": p["partyId"], "party_name": p["customer"],
                                  "recipient": p.get("recipient"), "subject": p["subject"], "body": body, "channel": channel,
                                  "amount": p.get("totalOverdue"), "references": p.get("invoices")})


# ----------------------------------------------------------------------------- approver edits

def edit_fields(action: PlannedAction) -> list[EditField]:
    if action.kind == "PLACE_HOLD":
        return [EditField("reason", "Reason recorded with the hold", "textarea", action.payload.get("reason"))]
    return [EditField("subject", "Subject", "text", action.payload.get("subject")),
            EditField("body", "Letter", "textarea", action.payload.get("body"),
                      help="The statement of invoices was generated from the ERP - keep the amounts as they are.")]


def apply_edits(action: PlannedAction, form: dict[str, Any]) -> dict[str, Any]:
    payload = dict(action.payload)
    if action.kind == "PLACE_HOLD":
        reason = (form.get("reason") or "").strip()
        if not reason:
            raise EditError("Give a reason for the account hold.")
        payload["reason"] = reason[:480]
        return payload
    subject = (form.get("subject") or "").strip()
    body = (form.get("body") or "").replace("\r\n", "\n").strip()
    if not subject or not body:
        raise EditError("The letter needs a subject and a text.")
    payload["subject"], payload["body"] = subject[:200], body[:20000]
    return payload


def settings(cfg) -> list[tuple[str, str]]:
    return [("Letters per run", f"up to {max_customers(cfg)} customers"),
            ("Contact interval", f"at most one letter per customer every {resend_days(cfg)} days"),
            ("Automatic sending", "reminders and second notices" if auto_send(cfg) else "off - every letter is approved"),
            ("Account hold", f"{hold_days(cfg)}+ days and at least {fmt(hold_min(cfg))} overdue")]

@TOOLS.tool("letters_sent", "AR", "Collections letters this AI employee already sent to the customer (newest first).",
            {"party_id": ID}, ("party_id",))
def letters_sent(ctx: ToolContext, party_id):
    pid = as_int(party_id, "party_id", True)
