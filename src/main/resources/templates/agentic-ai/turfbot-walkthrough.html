PKCE verifier, challenge, nonce, and state

"""PKCE helpers. The code_verifier and nonce travel inside the signed, time-limited `state` parameter,
so no server-side or cookie storage is needed between /login and /callback."""
from __future__ import annotations

import base64
import hashlib
import secrets
from urllib.parse import urlparse

from itsdangerous import BadSignature, SignatureExpired, URLSafeTimedSerializer

STATE_SALT = "turfbot-pkce"
STATE_MAX_AGE_SECONDS = 600


class StateError(ValueError):
    pass


def new_verifier() -> str:
    return secrets.token_urlsafe(64)[:96]  # 43..128 characters of the unreserved alphabet


def challenge(verifier: str) -> str:
    digest = hashlib.sha256(verifier.encode("ascii")).digest()
    return base64.urlsafe_b64encode(digest).rstrip(b"=").decode("ascii")


def safe_next(target: str | None) -> str:
    """Only same-site relative paths, never //host or absolute URLs (open-redirect guard)."""
    if not target:
        return "/"
    parsed = urlparse(target)
    if parsed.scheme or parsed.netloc or not target.startswith("/") or target.startswith("//") or "\\" in target:
        return "/"
    return target


def dump_state(secret_key: str, verifier: str, nonce: str, next_url: str) -> str:
    return URLSafeTimedSerializer(secret_key, salt=STATE_SALT).dumps({"cv": verifier, "n": nonce, "next": safe_next(next_url)})


def load_state(secret_key: str, state: str | None, max_age: int = STATE_MAX_AGE_SECONDS) -> dict:
    if not state:
        raise StateError("The sign-in response has no state parameter.")
    try:
        data = URLSafeTimedSerializer(secret_key, salt=STATE_SALT).loads(state, max_age=max_age)
    except SignatureExpired as e:
        raise StateError("The sign-in took longer than 10 minutes. Please sign in again.") from e
    except BadSignature as e:
        raise StateError("The sign-in response was not issued by TurfBot.") from e
    if not isinstance(data, dict) or not data.get("cv") or not data.get("n"):
        raise StateError("The sign-in state is incomplete.")
    data["next"] = safe_next(data.get("next"))
    return data

def _cfg():

def _redirect_uri() -> str:
    return f"{_cfg().BASE_URL.rstrip('/')}{url_for('auth.callback')}"


def _fail(title: str, message: str, code: int = 400):
    return render_template("error.html", code=code, title=title, message=message, show_login=True), code


@bp.get("/login")
def login():
    if current_user() is not None:
        return redirect(pkce.safe_next(request.args.get("next") or "/chat"))
    cfg = _cfg()
    verifier = pkce.new_verifier()
    nonce = secrets.token_urlsafe(24)
    state = pkce.dump_state(cfg.SECRET_KEY, verifier, nonce, request.args.get("next") or "/chat")
    try:
        authorize = keycloak().endpoint("authorization_endpoint")
    except KeycloakError as e:
        return _fail("Keycloak is unavailable", str(e), 503)
    query = urlencode({"response_type": "code", "client_id": cfg.KEYCLOAK_CLIENT_ID, "redirect_uri": _redirect_uri(),
                       "scope": "openid profile email", "state": state, "nonce": nonce,
                       "code_challenge": pkce.challenge(verifier), "code_challenge_method": "S256"})
    return redirect(f"{authorize}?{query}")


@bp.get("/callback")
def callback():
    cfg = _cfg()
    if request.args.get("error"):
        return _fail("Sign-in was not completed", request.args.get("error_description") or request.args["error"])
    try:
        state = pkce.load_state(cfg.SECRET_KEY, request.args.get("state"))
    except pkce.StateError as e:
        return _fail("Sign-in could not be verified", str(e))
    code = request.args.get("code")
    if not code:
        return _fail("Sign-in could not be verified", "Keycloak did not return an authorization code.")
    meta = keycloak()
    try:
        resp = requests.post(meta.endpoint("token_endpoint"), timeout=cfg.HTTP_TIMEOUT, data={
            "grant_type": "authorization_code", "code": code, "redirect_uri": _redirect_uri(),
            "client_id": cfg.KEYCLOAK_CLIENT_ID, "code_verifier": state["cv"]})
    except (requests.RequestException, KeycloakError) as e:
        return _fail("Keycloak is unavailable", f"The token exchange failed: {e}", 503)
    if resp.status_code != 200:
        log.warning("Token exchange refused: %s %s", resp.status_code, resp.text[:300])
        return _fail("Sign-in could not be verified", "Keycloak refused the authorization code. Please sign in again.")
    tokens = resp.json()
    try:
        id_claims = meta.verify(tokens["id_token"], audience=cfg.KEYCLOAK_CLIENT_ID, nonce=state["n"])
        access_claims = meta.verify(tokens["access_token"])
    except (KeyError, KeycloakError) as e:
        return _fail("Sign-in could not be verified", str(e))
    user = User(username=access_claims.get("preferred_username") or id_claims.get("preferred_username") or id_claims["sub"],
                name=id_claims.get("name") or id_claims.get("preferred_username") or "",
                email=id_claims.get("email"),
                roles=roles_from_claims(access_claims))
    session.clear()
    session.permanent = True
    session["user"] = user.to_session()
    session["id_token"] = tokens["id_token"]
    log.info("Signed in %s with roles %s", user.username, sorted(user.roles))
    return redirect(state["next"])

Authenticated session and role handling

"""The signed-in user, kept in Flask's signed session cookie, and the access decorators."""
from __future__ import annotations

import functools
from dataclasses import dataclass, field
from typing import Any, Callable

from flask import abort, jsonify, redirect, request, session, url_for


@dataclass(frozen=True)
class User:
    username: str
    name: str
    email: str | None
    roles: frozenset[str] = field(default_factory=frozenset)

    def has_any(self, *roles: str) -> bool:
        return any(r in self.roles for r in roles)

    @property
    def is_admin(self) -> bool:
        return "TC_ADMIN" in self.roles

    @property
    def is_auditor_or_admin(self) -> bool:
        return self.has_any("TC_ADMIN", "TC_AUDITOR")

    def to_session(self) -> dict[str, Any]:
        return {"username": self.username, "name": self.name, "email": self.email, "roles": sorted(self.roles)}

    @staticmethod
    def from_session(data: dict[str, Any] | None) -> "User | None":
        if not data or not data.get("username"):
            return None
        return User(data["username"], data.get("name") or data["username"], data.get("email"), frozenset(data.get("roles") or []))


ERP_ROLE_PREFIXES = ("CORE_", "GL_", "HORSE_", "PARTICIPANT_", "RACING_", "WAGERING_", "PRIZE_", "MEMBERSHIP_",
                     "HOSPITALITY_", "INVENTORY_", "PROCUREMENT_", "AP_", "AR_", "TREASURY_", "TC_")


def roles_from_claims(claims: dict[str, Any]) -> frozenset[str]:
    """The realm's flat `roles` claim (realm-roles-flat mapper), falling back to realm_access.roles.
    Keycloak expands composites, so TC_ADMIN arrives with every MANAGE and VIEW role."""
    raw = claims.get("roles")
    if not isinstance(raw, list):
        raw = (claims.get("realm_access") or {}).get("roles") or []
    return frozenset(r for r in raw if isinstance(r, str) and r.startswith(ERP_ROLE_PREFIXES))


def current_user() -> User | None:
    return User.from_session(session.get("user"))


def login_required(view: Callable) -> Callable:
    @functools.wraps(view)
    def wrapper(*args, **kwargs):
        if current_user() is None:
            if request.path.startswith("/api/"):
                return jsonify(error="unauthenticated", message="Your session has ended. Please sign in again."), 401
            return redirect(url_for("auth.login", next=request.full_path.rstrip("?")))
        return view(*args, **kwargs)
    return wrapper


def roles_required(*roles: str) -> Callable:
    def decorator(view: Callable) -> Callable:
        @functools.wraps(view)
        @login_required
        def wrapper(*args, **kwargs):
            if not current_user().has_any(*roles):
                abort(403)
            return view(*args, **kwargs)
        return wrapper
    return decorator

@bp.post("/api/chat")
@login_required
def chat():
    cfg = _cfg()
    user = current_user()
    body = request.get_json(silent=True) or {}
    question = str(body.get("message") or "").strip()
    if not question:
        return jsonify(error="empty", message="Type a question first."), 400
    if len(question) > cfg.QUESTION_MAX_CHARS:
        return jsonify(error="too_long", message=f"Questions are limited to {cfg.QUESTION_MAX_CHARS} characters."), 400
    if not cfg.ANTHROPIC_API_KEY and "agent_model_factory" not in current_app.config:
        return jsonify(error="no_api_key", message="ANTHROPIC_API_KEY is not set in the chatbot's .env file."), 503

    conversation_id = body.get("conversation_id")
    if conversation_id:
        conv = _own_conversation(int(conversation_id))
        conversation_id = conv["id"]
        history = store.history(conversation_id, cfg.HISTORY_TURNS)
    else:
        conversation_id = store.create_conversation(user, question.replace("\n", " ")[:80], cfg.ANTHROPIC_MODEL)
        history = []
    user_message_id = store.add_message(conversation_id, "USER", question)
    store.touch(conversation_id)

    tools = tools_for(user.roles)
    now = _club_now()
    ctx = ToolContext(api=api_client(), cfg=cfg, user=user, today=now.date(),
                      request_id=f"turfbot-{conversation_id}-{user_message_id}")
    recorder = store.DbRecorder(cfg, conversation_id, user.username)
    run = AgentRun(model=_model(), tools=tools, ctx=ctx, recorder=recorder,
                   request_context=request_context(user, _modules(tools), cfg.CLUB_TIMEZONE, now),
                   max_tool_rounds=cfg.MAX_TOOL_ROUNDS, prompt_cache=cfg.ANTHROPIC_PROMPT_CACHE)
    messages = [*history, HumanMessage(content=question)]
    title = store.conversation(conversation_id)["title"]

    @stream_with_context
    def generate():
        started = time.monotonic()
        saved = False
        try:
            yield _sse("conversation", {"id": conversation_id, "title": title, "questionId": user_message_id})
            for event in run_agent(run, messages):
                if event["event"] != "final":
                    yield _sse(event["event"], event)
                    continue
                latency = int((time.monotonic() - started) * 1000)
                message_id = store.add_message(conversation_id, "ASSISTANT", event["text"], event.get("stop_reason"),
                                               "tool budget reached" if event.get("budget_hit") else None, latency)
                recorder.attach(message_id)
                saved = True
                usage = run.usage_total
                yield _sse("done", {"messageId": message_id, "text": event["text"], "ms": latency,
                                    "rounds": event.get("rounds"), "usage": usage.as_dict(),
                                    "costUsd": float(usage.cost(cfg))})
        except Exception as exc:  # noqa: BLE001
            log.exception("Chat turn failed for %s (conversation %s)", user.username, conversation_id)
            friendly = errors.describe(exc, cfg.ANTHROPIC_MODEL)
            if not saved:
                message_id = store.add_message(conversation_id, "ASSISTANT", friendly, None, f"{exc.__class__.__name__}: {exc}",
                                               int((time.monotonic() - started) * 1000))
                recorder.attach(message_id)
                saved = True
            yield _sse("error", {"message": friendly})
        finally:
            if not saved:  # the browser went away mid-answer
                message_id = store.add_message(conversation_id, "ASSISTANT", "_The answer was interrupted._", None,
                                               "client disconnected", int((time.monotonic() - started) * 1000))
                recorder.attach(message_id)

    return Response(generate(), mimetype="text/event-stream",
                    headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"})

LangGraph agent/tools transitions and model calls

"""The agent graph:  START -> agent -> (tool calls? -> tools -> agent) -> END.

* agent: Claude (ChatAnthropic, Anthropic Messages API) bound to ONLY the tools the user's roles allow, streaming
  text to the browser as it is generated. The tool definitions and the stable system block carry cache_control
  breakpoints, so repeat calls (each tool round, each follow-up question) read them from the prompt cache.
* tools: runs every tool_use block of the last answer (role check again, arguments, API call), records each call,
  and returns tool_result blocks (is_error for failures, so Claude can recover or explain).
The loop stops after MAX_TOOL_ROUNDS tool rounds: the last agent call is told to answer from what it has.
"""
from __future__ import annotations

import logging
import time
from dataclasses import dataclass, field

EPHEMERAL = {"type": "ephemeral"}
BUDGET_NOTE = ("You have used every tool call allowed for this question. Do not call tools again: answer now from the "

class AgentState(TypedDict):
    messages: Annotated[list[AnyMessage], add_messages]
    rounds: int


class Recorder(Protocol):
    def tool_call(self, outcome: ToolOutcome) -> int | None: ...

    def usage(self, round_no: int, usage: Usage, stop_reason: str | None, request_id: str | None, latency_ms: int) -> None: ...


@dataclass
class AgentRun:
    model: Any                       # a LangChain chat model with bind_tools() / stream() - ChatAnthropic in production
    tools: list[ToolSpec]
    ctx: ToolContext
    recorder: Recorder
    request_context: str
    max_tool_rounds: int = 8
    prompt_cache: bool = True
    usage_total: Usage = field(default_factory=Usage)


def anthropic_tools(tools: list[ToolSpec], cache: bool) -> list[dict[str, Any]]:
    schemas = [t.anthropic_schema() for t in sorted(tools, key=lambda t: t.name)]  # stable order = stable cache key
    if cache and schemas:
        schemas[-1] = {**schemas[-1], "cache_control": EPHEMERAL}
    return schemas


def system_message(run: AgentRun, budget_spent: bool) -> SystemMessage:
    stable = {"type": "text", "text": STABLE_INSTRUCTIONS}
    if run.prompt_cache:
        stable["cache_control"] = EPHEMERAL
    tail = run.request_context + ("\n\n" + BUDGET_NOTE if budget_spent else "")
    return SystemMessage(content=[stable, {"type": "text", "text": tail}])


def build_graph(run: AgentRun):
    allowed = {t.name: t for t in run.tools}
    bound = run.model.bind_tools(anthropic_tools(run.tools, run.prompt_cache)) if run.tools else run.model

    def agent(state: AgentState) -> dict[str, Any]:
        writer = get_stream_writer()
        budget_spent = state.get("rounds", 0) >= run.max_tool_rounds
        messages = [system_message(run, budget_spent), *state["messages"]]
        started = time.monotonic()
        merged = None
        for chunk in bound.stream(messages):
            text = chunk.text
            if text:
                writer({"event": "token", "text": text})
            merged = chunk if merged is None else merged + chunk
        if merged is None:
            raise RuntimeError("The model returned an empty stream")
        message = message_chunk_to_message(merged)
        latency = int((time.monotonic() - started) * 1000)
        usage = Usage.from_message(message)
        run.usage_total = run.usage_total + usage
        meta = getattr(message, "response_metadata", {}) or {}
        run.recorder.usage(state.get("rounds", 0) + 1, usage, meta.get("stop_reason"), message.id, latency)
        if message.tool_calls:
            writer({"event": "thinking", "text": ""})  # separates streamed preamble text from the tool chips
        return {"messages": [message]}

    def tools(state: AgentState) -> dict[str, Any]:
        writer = get_stream_writer()
        last = state["messages"][-1]
        results: list[ToolMessage] = []
        for call in last.tool_calls:
            spec = allowed.get(call["name"])
            writer({"event": "tool_start", "id": call["id"], "name": call["name"], "args": call.get("args") or {},
                    "module": spec.module if spec else None})
            outcome = run_tool(call, allowed, run.ctx)
            db_id = run.recorder.tool_call(outcome)
            writer({"event": "tool_end", "id": call["id"], "name": call["name"], "status": outcome.status,
                    "rows": outcome.rows, "ms": outcome.duration_ms, "truncated": outcome.truncated,
                    "toolCallId": db_id, "error": outcome.error if outcome.status != "OK" else None})
            results.append(ToolMessage(content=outcome.model_text, tool_call_id=call["id"], name=call["name"],
                                       status="success" if outcome.status == "OK" else "error"))
        return {"messages": results, "rounds": state.get("rounds", 0) + 1}

    def route(state: AgentState) -> str:
        last = state["messages"][-1]
        if isinstance(last, AIMessage) and last.tool_calls and state.get("rounds", 0) < run.max_tool_rounds:
            return "tools"
        return END

    graph = StateGraph(AgentState)
    graph.add_node("agent", agent)
    graph.add_node("tools", tools)
    graph.add_edge(START, "agent")
    graph.add_conditional_edges("agent", route, {"tools": "tools", END: END})
    graph.add_edge("tools", "agent")
    return graph.compile()


def final_text(message: AnyMessage | None) -> str:
    if message is None:
        return ""
    return (message.text or "").strip()


def run_agent(run: AgentRun, history: list[AnyMessage]) -> Iterator[dict[str, Any]]:
    """Streams {"event": token | thinking | tool_start | tool_end} dicts, then one {"event": "final"}."""
    graph = build_graph(run)
    state: dict[str, Any] | None = None
    for mode, chunk in graph.stream({"messages": history, "rounds": 0}, stream_mode=["custom", "values"],
                                    config={"recursion_limit": 2 * run.max_tool_rounds + 6}):
        if mode == "custom":
            yield chunk
        else:
            state = chunk
    messages = (state or {}).get("messages", [])
    last = messages[-1] if messages else None
    stop_reason = (getattr(last, "response_metadata", {}) or {}).get("stop_reason") if last is not None else None
    budget_hit = isinstance(last, AIMessage) and bool(last.tool_calls)
    text = final_text(last) if isinstance(last, AIMessage) else ""
    if budget_hit:
        text = (text + "\n\n" if text else "") + ("_I could not finish this within the allowed number of lookups. "
                                                  "Try a narrower question._")
    if not text:
        text = "_No answer was produced._"
    yield {"event": "final", "text": text, "stop_reason": stop_reason, "budget_hit": budget_hit,
           "rounds": (state or {}).get("rounds", 0)}

@dataclass
class ToolOutcome:
    tool_use_id: str
    name: str
    module: str
    arguments: dict[str, Any]
    status: str                  # OK / DENIED / ERROR
    rows: int | None
    model_text: str              # what the model sees (truncated JSON or an error sentence)
    stored_json: str | None      # full result kept for the Excel export
    error: str | None
    duration_ms: int
    truncated: bool = False


def _dumps(value: Any) -> str:
    return json.dumps(value, ensure_ascii=False, default=str, separators=(",", ":"))


def count_rows(result: Any) -> int:
    if isinstance(result, list):
        return len(result)
    if isinstance(result, dict):
        lists = [len(v) for v in result.values() if isinstance(v, list) and v and isinstance(v[0], dict)]
        if lists:
            return max(lists)
        return 0 if result.get("found") is False and not result.get("candidates") else 1
    return 0 if result is None else 1


def _cut(value: Any, max_rows: int) -> tuple[Any, bool]:
    """Cuts every list longer than max_rows (recursively)."""
    if isinstance(value, list):
        cut = len(value) > max_rows
        items = []
        for v in value[:max_rows]:
            nv, c = _cut(v, max_rows)
            items.append(nv)
            cut = cut or c
        return items, cut
    if isinstance(value, dict):
        out, cut = {}, False
        for k, v in value.items():
            out[k], c = _cut(v, max_rows)
            cut = cut or c
        return out, cut
    return value, False


def shape_for_model(result: Any, max_rows: int, max_chars: int) -> tuple[str, bool]:
    total = count_rows(result)
    rows = max_rows
    while True:
        shaped, cut = _cut(result, rows)
        if cut:
            shaped = {"truncated": True, "rowCount": total,
                      "note": f"Only the first {rows} rows of each list are included. Tell the user and suggest a narrower question.",
                      "data": shaped}
        text = _dumps(shaped)
        if len(text) <= max_chars or rows <= 5:
            break
        rows = max(rows // 2, 5)
    if len(text) > max_chars:
        return text[:max_chars] + ' ...[cut: the result is too large; ask a narrower question]', True
    return text, cut


def run_tool(call: dict[str, Any], allowed: dict[str, ToolSpec], ctx: ToolContext) -> ToolOutcome:
    name = call.get("name") or ""
    args = call.get("args") or {}
    tool_use_id = call.get("id") or ""
    spec = REGISTRY.get(name)
    started = time.monotonic()

    def outcome(status, text, rows=None, stored=None, error=None, truncated=False):
        return ToolOutcome(tool_use_id, name, spec.module if spec else "UNKNOWN", args, status, rows, text, stored, error,
                           int((time.monotonic() - started) * 1000), truncated)

    if spec is None:
        return outcome("ERROR", f"Unknown tool '{name}'.", error="unknown tool")
    if name not in allowed or not spec.allowed(ctx.user.roles):
        log.warning("DENIED tool %s for %s", name, ctx.user.username)
        return outcome("DENIED", f"Access denied: {ctx.user.name}'s roles do not include the "
                                 f"{ctx.cfg.module_name(spec.module) if spec.module in _modules() else spec.module} module. "
                                 f"Tell the user this data is outside their access.", error="role check failed")
    params = inspect.signature(spec.fn).parameters
    unknown = [k for k in args if k not in params]
    if unknown:
        return outcome("ERROR", f"Unknown argument(s) {unknown} for {name}; allowed: {list(spec.properties)}.",
                       error=f"unknown arguments {unknown}")
    try:
        result = spec.fn(ctx, **args)
    except ToolInputError as e:
        return outcome("ERROR", f"Invalid input: {e}", error=str(e))
    except ApiError as e:
        return outcome("ERROR", f"The lookup failed: {e}", error=str(e))
    except TypeError as e:
        return outcome("ERROR", f"Invalid arguments for {name}: {e}", error=str(e))
    except Exception as e:  # noqa: BLE001 - a broken tool must not break the chat
        log.exception("Tool %s failed", name)
        return outcome("ERROR", f"The lookup failed with an internal error ({e.__class__.__name__}).", error=repr(e)[:900])

    text, truncated = shape_for_model(result, ctx.cfg.TOOL_RESULT_MAX_ROWS, ctx.cfg.TOOL_RESULT_MAX_CHARS)
    stored = _dumps(result)
    if len(stored) > ctx.cfg.STORED_RESULT_MAX_CHARS:
        stored = _dumps({"truncated": True, "note": "result too large to keep", "preview": text})
    return outcome("OK", text, rows=count_rows(result), stored=stored, truncated=truncated)

"""TurfBot's tools: read-only calls to the ERP services' /api/v1 endpoints.

Every tool belongs to one ERP module. A user is offered a tool only when their Keycloak roles would let them open
that module in the UI (<MODULE>_VIEW or <MODULE>_MANAGE; TC_ADMIN / TC_AUDITOR see every module), and the audit
tools only for TC_AUDITOR / TC_ADMIN - the same rule as core-service's audit pages. The tools node checks again
before running anything, so a tool the model was not given is refused (logged as DENIED) rather than executed.

The services are called as svc-chatbot (SVC_READ), so the role filter here is what keeps the answer inside the
user's own access.

@dataclass(frozen=True)
class ToolSpec:
    name: str
    module: str
    description: str
    fn: Callable[..., Any]
    properties: dict[str, Any] = field(default_factory=dict)
    required: tuple[str, ...] = ()
    roles: tuple[str, ...] | None = None  # None = the module's VIEW / MANAGE role

    def allowed(self, roles: frozenset[str] | set[str]) -> bool:
        if self.roles is not None:
            return any(r in roles for r in self.roles)
        return bool({f"{self.module}_VIEW", f"{self.module}_MANAGE", "TC_ADMIN", "TC_AUDITOR"} & set(roles))

    def anthropic_schema(self) -> dict[str, Any]:
        """The Messages API tool format: name, description, input_schema (JSON Schema)."""
        return {"name": self.name, "description": self.description,
                "input_schema": {"type": "object", "properties": self.properties, "required": list(self.required)}}


REGISTRY: dict[str, ToolSpec] = {}
AUDIT_ROLES = ("TC_AUDITOR", "TC_ADMIN")


def tool(name: str, module: str, description: str, properties: dict | None = None, required: tuple[str, ...] = (),
         roles: tuple[str, ...] | None = None):
    def decorator(fn):
        if name in REGISTRY:
            raise RuntimeError(f"Duplicate tool {name}")
        REGISTRY[name] = ToolSpec(name, module, description.strip(), fn, properties or {}, required, roles)
        return fn
    return decorator


def tools_for(roles) -> list[ToolSpec]:
    return [t for t in REGISTRY.values() if t.allowed(roles)]

def _resolve_party(ctx: ToolContext, party: Any) -> dict[str, Any]:
    text = str(party or "").strip()
    if not text:
        raise ToolInputError("party is required")
    if text.isdigit():
        p = ctx.get("CORE", f"/parties/{int(text)}")
        return {**p, "found": True}
    hits = _content(ctx.get("CORE", "/parties", q=text, size=10))
    if not hits:
        # the API matches the whole phrase; retry with the longest word ("Blue Ridge's" -> "Ridge")
        words = sorted(_norm(text).split(), key=len, reverse=True)
        if words and words[0] != _norm(text):
            hits = [h for h in _content(ctx.get("CORE", "/parties", q=words[0], size=25))
                    if _matches(text, h.get("displayName"), h.get("externalRef"))]
    if not hits:
        return {"found": False, "message": f"No party matches '{text}'."}
    exact = [h for h in hits if _norm(h.get("displayName")) == _norm(text) or _norm(h.get("externalRef")) == _norm(text)]
    if len(exact) == 1:
        return {**exact[0], "found": True}
    if len(hits) == 1:
        return {**hits[0], "found": True}
    return {"found": False, "candidates": [{"id": h["id"], "name": h["displayName"], "type": h.get("type"),
                                            "externalRef": h.get("externalRef")} for h in hits],
            "message": "Several parties match - ask the user which one they mean."}

"""GET client for the ERP services' /api/v1 endpoints, authenticated as svc-chatbot."""
from __future__ import annotations

import logging
import uuid
from typing import Any

import requests

from .keycloak import KeycloakError

log = logging.getLogger(__name__)


class ApiError(RuntimeError):
    """A failure the model can explain to the user (service down, not found, refused)."""

    def __init__(self, message: str, status: int | None = None):
        super().__init__(message)
        self.status = status


class ApiClient:
    def __init__(self, cfg, tokens):
        self.cfg = cfg
        self.tokens = tokens
        self.session = requests.Session()

    def get(self, module: str, path: str, params: dict[str, Any] | None = None, request_id: str | None = None) -> Any:
        base = self.cfg.module_url(module)
        url = f"{base}/api/v1{path}"
        clean = {k: v for k, v in (params or {}).items() if v not in (None, "")}
        rid = request_id or f"turfbot-{uuid.uuid4().hex[:16]}"
        name = self.cfg.module_name(module)
        for attempt in (1, 2):
            try:
                token = self.tokens.token(force_refresh=attempt == 2)
            except KeycloakError as e:
                raise ApiError(str(e)) from e
            try:
                resp = self.session.get(url, params=clean, timeout=self.cfg.HTTP_TIMEOUT,
                                        headers={"Authorization": f"Bearer {token}", "Accept": "application/json",
                                                 "X-Request-Id": rid})
            except requests.ConnectionError as e:
                raise ApiError(f"The {name} service is not running or not reachable at {base}.") from e
            except requests.Timeout as e:
                raise ApiError(f"The {name} service did not answer within {self.cfg.HTTP_TIMEOUT} seconds.") from e
            if resp.status_code == 401 and attempt == 1:
                continue  # token revoked or realm restarted: fetch a new one once
            break
        log.info("GET %s %s -> %s", module, resp.url, resp.status_code)
        if resp.status_code == 200:
            return resp.json() if resp.content else None
        if resp.status_code == 404:
            raise ApiError(f"{name}: not found ({_detail(resp) or path}).", 404)
        if resp.status_code in (401, 403):
            raise ApiError(f"{name} refused the chatbot's service account (HTTP {resp.status_code}). "
                           f"Check that svc-chatbot has SVC_READ in Keycloak.", resp.status_code)
        if resp.status_code == 400:
            raise ApiError(f"{name} rejected the request: {_detail(resp) or 'bad parameters'}.", 400)

class KeycloakError(RuntimeError):
    pass


class KeycloakMetadata:
    """Caches the realm's /.well-known/openid-configuration for an hour and verifies tokens against its JWKS."""

    TTL_SECONDS = 3600

    def __init__(self, issuer: str, timeout: int = 15):
        self.issuer = issuer.rstrip("/")
        self.timeout = timeout
        self._doc: dict[str, Any] | None = None
        self._loaded_at = 0.0
        self._jwks: jwt.PyJWKClient | None = None
        self._lock = threading.Lock()

    def document(self) -> dict[str, Any]:
        with self._lock:
            if self._doc is None or time.time() - self._loaded_at > self.TTL_SECONDS:
                url = f"{self.issuer}/.well-known/openid-configuration"
                try:
                    resp = requests.get(url, timeout=self.timeout)
                    resp.raise_for_status()
                except requests.RequestException as e:
                    raise KeycloakError(f"Keycloak is not reachable at {url} ({e.__class__.__name__})") from e
                self._doc = resp.json()
                self._loaded_at = time.time()
                self._jwks = jwt.PyJWKClient(self._doc["jwks_uri"], cache_keys=True, lifespan=self.TTL_SECONDS,
                                             timeout=self.timeout)
            return self._doc

    def endpoint(self, name: str) -> str:
        return self.document()[name]

    def verify(self, token: str, audience: str | None = None, nonce: str | None = None) -> dict[str, Any]:
        """Verifies signature, issuer, expiry (and audience / nonce when given) and returns the claims."""
        self.document()
        try:
            key = self._jwks.get_signing_key_from_jwt(token)
            claims = jwt.decode(token, key.key, algorithms=["RS256", "RS384", "RS512", "PS256", "ES256"],
                                audience=audience, issuer=self.issuer, leeway=30,
                                options={"verify_aud": audience is not None, "require": ["exp", "iat", "iss"]})
        except jwt.PyJWTError as e:
            raise KeycloakError(f"Token rejected: {e}") from e
        if nonce is not None and claims.get("nonce") != nonce:
            raise KeycloakError("Token rejected: nonce mismatch")
        return claims


class ServiceTokenProvider:
    """svc-chatbot access token (client credentials), cached until 30 seconds before it expires. Thread-safe."""

    def __init__(self, metadata: KeycloakMetadata, client_id: str, client_secret: str, timeout: int = 15):
        self.metadata = metadata
        self.client_id = client_id
        self.client_secret = client_secret
        self.timeout = timeout
        self._token: str | None = None
        self._expires_at = 0.0
        self._lock = threading.Lock()

    def token(self, force_refresh: bool = False) -> str:
        with self._lock:
            if force_refresh or self._token is None or time.time() >= self._expires_at:
                url = self.metadata.endpoint("token_endpoint")
                try:
                    resp = requests.post(url, data={"grant_type": "client_credentials"},
                                         auth=(self.client_id, self.client_secret), timeout=self.timeout)
                except requests.RequestException as e:
                    raise KeycloakError(f"Keycloak is not reachable at {url} ({e.__class__.__name__})") from e
                if resp.status_code != 200:
                    raise KeycloakError(f"Keycloak refused the {self.client_id} credentials (HTTP {resp.status_code}): "
                                        f"{resp.text[:200]}")
                body = resp.json()
                self._token = body["access_token"]
                self._expires_at = time.time() + max(int(body.get("expires_in", 300)) - 30, 10)
                log.debug("Fetched %s service token (expires in %ss)", self.client_id, body.get("expires_in"))
            return self._token

"""Token usage and cost of one Messages API call, from LangChain's usage_metadata."""
from __future__ import annotations

from dataclasses import dataclass
from decimal import ROUND_HALF_UP, Decimal
from typing import Any

MTOK = Decimal(1_000_000)


@dataclass(frozen=True)
class Usage:
    input_tokens: int = 0            # uncached input
    output_tokens: int = 0
    cache_creation_tokens: int = 0
    cache_read_tokens: int = 0

    @staticmethod
    def from_message(message: Any) -> "Usage":
        meta = getattr(message, "usage_metadata", None) or {}
        details = meta.get("input_token_details") or {}
        cache_read = int(details.get("cache_read") or 0)
        # newer responses split cache writes by TTL (ephemeral_5m / ephemeral_1h) and LangChain then zeroes "cache_creation"
        cache_write = int(details.get("cache_creation") or 0) or (
            int(details.get("ephemeral_5m_input_tokens") or 0) + int(details.get("ephemeral_1h_input_tokens") or 0))
        # LangChain reports input_tokens as the total prompt (uncached + cache write + cache read)
        total_in = int(meta.get("input_tokens") or 0)
        return Usage(max(total_in - cache_read - cache_write, 0), int(meta.get("output_tokens") or 0), cache_write, cache_read)

    def cost(self, cfg) -> Decimal:
        value = (Decimal(self.input_tokens) * cfg.PRICE_INPUT_PER_MTOK
                 + Decimal(self.output_tokens) * cfg.PRICE_OUTPUT_PER_MTOK
                 + Decimal(self.cache_creation_tokens) * cfg.PRICE_CACHE_WRITE_PER_MTOK
                 + Decimal(self.cache_read_tokens) * cfg.PRICE_CACHE_READ_PER_MTOK) / MTOK
        return value.quantize(Decimal("0.000001"), rounding=ROUND_HALF_UP)

    def __add__(self, other: "Usage") -> "Usage":
        return Usage(self.input_tokens + other.input_tokens, self.output_tokens + other.output_tokens,
                     self.cache_creation_tokens + other.cache_creation_tokens, self.cache_read_tokens + other.cache_read_tokens)

    def as_dict(self) -> dict[str, int]:
        return {"input_tokens": self.input_tokens, "output_tokens": self.output_tokens,
                "cache_creation_tokens": self.cache_creation_tokens, "cache_read_tokens": self.cache_read_tokens}
