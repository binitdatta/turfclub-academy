Replenishment process, decision flow and execution safeguards

"""AI Stores Replenisher - watches stock 24/7 and raises purchase requisitions for what is running out.

Work: every WATCH_MINUTES it checks inventory's reorder report; when an item drops to its reorder point and nothing
already covers it (an open requisition, an open purchase order or a requisition waiting in the workforce inbox), it
starts a run. The model looks at upcoming racedays and club events, recent prices and stock by location, and proposes
quantities. The policy then decides:
  * estimated total <= REPLENISHER_AUTO_LIMIT (default $750) and every line within bounds -> AUTO: the requisition is
    raised and submitted straight away (a buyer still approves and orders it in procurement, as for any requisition);
  * above the limit, a quantity far above the ERP's suggestion, a price far from standard cost, or no proposal at all
    -> APPROVAL by a buyer (PROCUREMENT_MANAGE) in the workforce inbox, who may change quantities first.
Write: POST /api/v1/requisitions as service-account-ai-replenisher (role SVC_REQUISITION_CREATE) - nothing else.
"""
from __future__ import annotations

import hashlib
from datetime import timedelta
from decimal import ROUND_HALF_UP, Decimal
from pathlib import Path
from typing import Any

from ...db import from_json
from ...platform.api import ApiError
from ..base import (APPROVAL, AUTO, SKIP, ActionResult, EditError, EditField, Employee, ExecContext, Facts,
                    PlannedAction)
from .tools import OPEN_PO_STATUSES, OPEN_REQ_STATUSES, TOOLS

CODE = "replenisher"
REQUESTER_REF = "AI-REPLENISHER"


def _d(value: Any) -> Decimal:
    try:
        return Decimal(str(value if value not in (None, "") else 0))
    except Exception:  # noqa: BLE001
        return Decimal(0)


def _money(value: Decimal) -> float:
    return float(value.quantize(Decimal("0.01"), rounding=ROUND_HALF_UP))


def _qty(value: Decimal) -> float:
    return float(value.quantize(Decimal("0.001"), rounding=ROUND_HALF_UP))


def auto_limit(cfg) -> Decimal:
    return cfg.employee_dec(CODE, "AUTO_LIMIT", "750")


def max_factor(cfg) -> Decimal:
    return cfg.employee_dec(CODE, "MAX_QTY_FACTOR", "3")


def price_tolerance(cfg) -> Decimal:
    return cfg.employee_dec(CODE, "PRICE_TOLERANCE_PCT", "25") / Decimal(100)


def lead_days(cfg) -> int:
    return cfg.employee_int(CODE, "LEAD_DAYS", 7)


# ----------------------------------------------------------------------------- prepare (deterministic, no LLM)

def covered_quantities(ctx) -> tuple[dict[int, Decimal], dict[int, Decimal], dict[int, Decimal]]:
    """Per item id: quantity on open requisitions, outstanding on open purchase orders, and waiting in the inbox."""
    requested: dict[int, Decimal] = {}
    for status in OPEN_REQ_STATUSES:
        for r in ctx.get("PROCUREMENT", "/requisitions", status=status):
            for line in r.get("lines") or []:
                requested[line["itemId"]] = requested.get(line["itemId"], Decimal(0)) + _d(line.get("quantity"))
    on_order: dict[int, Decimal] = {}
    for status in OPEN_PO_STATUSES:
        for po in ctx.get("PROCUREMENT", "/purchase-orders", status=status):
            detail = ctx.get("PROCUREMENT", f"/purchase-orders/{po['id']}")
            for line in detail.get("lines") or []:
                on_order[line["itemId"]] = on_order.get(line["itemId"], Decimal(0)) + _d(line.get("outstanding"))
    pending: dict[int, Decimal] = {}
    for row in ctx.db.query("SELECT payload, edited_payload FROM approval WHERE employee_code = %s AND status = 'PENDING'",
                            (CODE,)):
        payload = from_json(row.get("edited_payload")) or from_json(row.get("payload")) or {}
        for line in payload.get("lines") or []:
            pending[int(line["itemId"])] = pending.get(int(line["itemId"]), Decimal(0)) + _d(line.get("quantity"))
    return requested, on_order, pending


def requester_party(ctx) -> dict[str, Any]:
    page = ctx.get("CORE", "/parties", q="AI Stores Replenisher", size=25)
    for p in (page or {}).get("content") or []:
        if p.get("externalRef") == REQUESTER_REF:
            return p
    raise ApiError(f"The party register has no party with external ref {REQUESTER_REF}. Apply database/sql/32_workforce.sql.")


def prepare(ctx) -> Facts:
    below = ctx.get("INVENTORY", "/stock/below-reorder") or []
    if not below:
        return Facts(True, "No item is at or below its reorder point.", {"belowReorder": 0}, fingerprint="")
    requested, on_order, pending = covered_quantities(ctx)
    worklist, covered = [], []
    for r in below:
        item_id = int(r["itemId"])
        suggested = _d(r.get("suggested"))
        cover = requested.get(item_id, Decimal(0)) + on_order.get(item_id, Decimal(0)) + pending.get(item_id, Decimal(0))
        net = suggested - cover
        row = {"itemId": item_id, "sku": r.get("sku"), "description": r.get("description"), "uom": r.get("uom"),
               "category": r.get("category"), "onHand": _qty(_d(r.get("onHand"))), "reorderPoint": _qty(_d(r.get("reorderPoint"))),
               "erpSuggested": _qty(suggested), "onOpenRequisitions": _qty(requested.get(item_id, Decimal(0))),
               "onOpenPurchaseOrders": _qty(on_order.get(item_id, Decimal(0))),
               "waitingForApproval": _qty(pending.get(item_id, Decimal(0))), "netToOrder": _qty(max(net, Decimal(0))),
               "standardCost": float(_d(r.get("standardCost"))),
               "valueAtStandard": _money(max(net, Decimal(0)) * _d(r.get("standardCost")))}
        (worklist if net > 0 else covered).append(row)
    fingerprint = hashlib.sha1(",".join(f"{w['itemId']}:{w['netToOrder']}" for w in sorted(worklist, key=lambda x: x["itemId"]))
                               .encode()).hexdigest() if worklist else ""
    if not worklist:
        return Facts(True, f"{len(below)} item(s) are below their reorder point, but open requisitions and purchase orders "
                           f"already cover all of them.", {"belowReorder": len(below), "covered": covered}, fingerprint)
    party = requester_party(ctx)
    value = sum(Decimal(str(w["valueAtStandard"])) for w in worklist)
    headline = (f"{len(worklist)} item(s) need ordering (est. ${_money(value):,.2f} at standard cost); "
                f"{len(covered)} more are below reorder point but already covered.")
    return Facts(False, headline, {"asOf": ctx.today.isoformat(), "autoApprovalLimit": float(auto_limit(ctx.cfg)),
                                   "maxQuantityFactor": float(max_factor(ctx.cfg)), "leadDays": lead_days(ctx.cfg),
                                   "requesterPartyId": party["id"], "requester": party.get("displayName"),
                                   "worklist": worklist, "alreadyCovered": covered}, fingerprint)


# ----------------------------------------------------------------------------- plan (deterministic policy)

PROPOSAL_SCHEMA = {
    "type": "object",
    "properties": {
        "summary": {"type": "string", "description": "2-4 sentences for the buyer: what you are ordering and why (racedays, "
                                                     "events, price changes). Plain English, no markdown."},
        "lines": {"type": "array", "description": "One line per item to requisition. Only items from the worklist.",
                  "items": {"type": "object", "properties": {
                      "item_id": {"type": "integer"},
                      "quantity": {"type": "number", "description": "Quantity in the item's unit"},
                      "estimated_unit_cost": {"type": "number", "description": "Expected unit price; omit for the standard cost"},
                      "needed_by": {"type": "string", "format": "date", "description": "ISO date; omit for today + lead days"},
                      "reason": {"type": "string", "description": "One short sentence: why this quantity"}},
                      "required": ["item_id", "quantity", "reason"]}},
        "not_ordered": {"type": "array", "description": "Worklist items you decided NOT to order, with the reason.",
                        "items": {"type": "object", "properties": {"item_id": {"type": "integer"}, "reason": {"type": "string"}},
                                  "required": ["item_id", "reason"]}},
    },
    "required": ["summary", "lines"],
}


def plan(ctx, facts: Facts, proposal: dict | None) -> list[PlannedAction]:
    cfg = ctx.cfg
    work = {int(w["itemId"]): w for w in facts.data.get("worklist") or []}
    reasons: list[str] = []
    notes: list[str] = []
    needed_default = (ctx.today + timedelta(days=lead_days(cfg))).isoformat()
    if proposal is None:
        reasons.append("The AI did not produce a proposal, so the ERP's suggested quantities are used and a buyer must approve.")
        proposed = [{"item_id": i, "quantity": w["netToOrder"], "reason": "ERP suggested quantity"} for i, w in work.items()]
        summary = "Requisition at the ERP's suggested quantities for every uncovered item below its reorder point."
    else:
        proposed = proposal.get("lines") or []
        summary = (proposal.get("summary") or "").strip() or "Replenishment requisition."
    lines, seen = [], set()
    for p in proposed:
        try:
            item_id = int(p.get("item_id"))
        except (TypeError, ValueError):
            notes.append(f"Ignored a line with an invalid item id ({p.get('item_id')!r}).")
            continue
        w = work.get(item_id)
        if w is None:
            notes.append(f"Ignored item {item_id}: it is not on the worklist (not below reorder point or already covered).")
            continue
        if item_id in seen:
            notes.append(f"Ignored a duplicate line for {w['sku']}.")
            continue
        seen.add(item_id)
        qty = _d(p.get("quantity"))
        if qty <= 0:
            notes.append(f"Ignored {w['sku']}: quantity {qty} is not positive.")
            continue
        net = _d(w["netToOrder"])
        cap = max(net * max_factor(cfg), _d(w["reorderPoint"]), Decimal(1))
        if qty > cap:
            reasons.append(f"{w['sku']}: the AI asked for {_qty(qty)} {w.get('uom') or ''}, more than {max_factor(cfg)}x the "
                           f"ERP's net suggestion ({_qty(net)}) - capped at {_qty(cap)}.")
            qty = cap
        std = _d(w["standardCost"])
        cost = _d(p.get("estimated_unit_cost")) if p.get("estimated_unit_cost") not in (None, "") else std
        if cost < 0:
            cost = std
        if std > 0 and abs(cost - std) / std > price_tolerance(cfg):
            reasons.append(f"{w['sku']}: estimated unit cost ${float(cost):,.4f} is more than "
                           f"{int(price_tolerance(cfg) * 100)}% away from the standard cost ${float(std):,.4f}.")
        lines.append({"itemId": item_id, "sku": w["sku"], "description": w["description"], "uom": w.get("uom"),
                      "quantity": _qty(qty), "estimatedUnitCost": float(cost.quantize(Decimal("0.0001"))),
                      "neededBy": str(p.get("needed_by") or needed_default)[:10], "reason": (p.get("reason") or "").strip()[:300],
                      "standardCost": float(std), "netToOrder": w["netToOrder"],
                      "lineTotal": _money(qty * cost)})
    not_ordered = []
    for n in (proposal or {}).get("not_ordered") or []:
        try:
            w = work.get(int(n.get("item_id")))
        except (TypeError, ValueError):
            w = None
        if w is not None and w["itemId"] not in seen:
            not_ordered.append({"itemId": w["itemId"], "sku": w["sku"], "reason": (n.get("reason") or "").strip()[:300]})
    left_out = [w["sku"] for i, w in work.items() if i not in seen and i not in {x["itemId"] for x in not_ordered}]
    if left_out:
        notes.append(f"Left out without a reason: {', '.join(left_out)}.")
    total = sum((Decimal(str(line["lineTotal"])) for line in lines), Decimal(0))
    payload = {"requesterPartyId": facts.data.get("requesterPartyId"), "requester": facts.data.get("requester"),
               "requestedOn": ctx.today.isoformat(), "lines": lines, "notOrdered": not_ordered, "notes": notes,
               "estimatedTotal": _money(total)}
    if not lines:
        return [PlannedAction("requisition", "CREATE_REQUISITION", "No requisition needed", summary, payload, SKIP,
                              ["The AI decided that nothing on the worklist should be ordered now."] + notes, 0.0)]
    limit = auto_limit(cfg)
    if total > limit:
        reasons.insert(0, f"Estimated total ${_money(total):,.2f} is above the ${_money(limit):,.2f} auto-approval limit.")
    title = f"Requisition: {len(lines)} item{'s' if len(lines) != 1 else ''}, est. ${_money(total):,.2f}"
    return [PlannedAction("requisition", "CREATE_REQUISITION", title, summary, payload, APPROVAL if reasons else AUTO,
                          reasons + notes, _money(total))]


# ----------------------------------------------------------------------------- execute (deterministic ERP write)

def execute(ectx: ExecContext, action: PlannedAction) -> ActionResult:
    p = action.payload
    body = {"requesterPartyId": p["requesterPartyId"], "requestedOn": p.get("requestedOn"),
            "lines": [{"itemId": line["itemId"], "quantity": line["quantity"], "estimatedUnitCost": line["estimatedUnitCost"],
                       "neededBy": line.get("neededBy")} for line in p["lines"] if float(line["quantity"]) > 0]}
    try:
        created = ectx.api.post("PROCUREMENT", "/requisitions", body)
    except ApiError as e:
        return ActionResult("FAILED", error=str(e), response=e.body)
    return ActionResult("DONE", erp_reference=f"{created.get('requisitionNo')} ({created.get('status')})", response=created)


# ----------------------------------------------------------------------------- approver edits

def _plain(value: float) -> str:
    return f"{value:.3f}".rstrip("0").rstrip(".")


def edit_fields(action: PlannedAction) -> list[EditField]:
    fields = []
    for line in action.payload.get("lines") or []:
        fields.append(EditField(f"qty_{line['itemId']}", f"{line['sku']} - {line['description']} ({line.get('uom') or 'units'})",
                                "number", _plain(line["quantity"]), help=f"ERP net suggestion {_plain(line['netToOrder'])}; 0 removes the line",
                                min=0))
    return fields


def apply_edits(action: PlannedAction, form: dict[str, Any]) -> dict[str, Any]:
    payload = {**action.payload, "lines": []}
    changed = []
    for line in action.payload.get("lines") or []:
        raw = form.get(f"qty_{line['itemId']}")
        if raw in (None, ""):
            payload["lines"].append(line)
            continue
        try:
            qty = Decimal(str(raw))
        except Exception as e:  # noqa: BLE001
            raise EditError(f"{line['sku']}: '{raw}' is not a number.") from e
        if qty < 0:
            raise EditError(f"{line['sku']}: the quantity cannot be negative.")
        if qty == 0:
            changed.append(f"removed {line['sku']}")
            continue
        if _qty(qty) != line["quantity"]:
            changed.append(f"{line['sku']} {line['quantity']} -> {_qty(qty)}")
        payload["lines"].append({**line, "quantity": _qty(qty),
                                 "lineTotal": _money(qty * Decimal(str(line["estimatedUnitCost"])))})
    if not payload["lines"]:
        raise EditError("Every line was removed - reject the requisition instead.")
    payload["estimatedTotal"] = _money(sum(Decimal(str(x["lineTotal"])) for x in payload["lines"]))
    payload["edits"] = changed
    return payload


def settings(cfg) -> list[tuple[str, str]]:
    return [("Auto-approval limit", f"${_money(auto_limit(cfg)):,.2f}"),
            ("Max quantity", f"{max_factor(cfg)}x the ERP's net suggestion"),
            ("Price tolerance", f"{int(price_tolerance(cfg) * 100)}% from standard cost"),
            ("Default lead time", f"{lead_days(cfg)} days")]

"""Read tools of the AI Stores Replenisher (service account ai-replenisher: SVC_READ + SVC_REQUISITION_CREATE)."""
from __future__ import annotations

from datetime import date, datetime, timedelta

from ...platform.tools import ID, ToolContext, Toolbox, as_int, matches, s

TOOLS = Toolbox("replenisher")

OPEN_PO_STATUSES = ("DRAFT", "APPROVED", "SENT", "PART_RECEIVED")
OPEN_REQ_STATUSES = ("DRAFT", "SUBMITTED", "APPROVED")


@TOOLS.tool("stock_on_hand", "INVENTORY",
            "Quantity and value on hand of an item at every stock location. Use it to see where the stock sits.",
            {"item": s("string", "SKU or description fragment")}, ("item",))
def stock_on_hand(ctx: ToolContext, item):
    return [r for r in ctx.get("INVENTORY", "/stock") if matches(item, r.get("sku"), r.get("description"))]


@TOOLS.tool("item_details", "INVENTORY", "One stock item by id: SKU, description, category, unit, standard cost, reorder "
            "point, active flag.", {"item_id": ID}, ("item_id",))
def item_details(ctx: ToolContext, item_id):
    return ctx.get("INVENTORY", f"/items/{as_int(item_id, 'item_id', True)}")


@TOOLS.tool("price_history", "PROCUREMENT",
            "What the club recently paid for an item: the unit price, supplier and date on the latest purchase orders "
            "that contain it (newest first). Use it to set a realistic estimated unit cost.",
            {"sku": s("string", "The item's SKU"), "orders_to_scan": s("integer", "How many recent purchase orders to "
                                                                        "scan (default 25, max 60)")}, ("sku",))
def price_history(ctx: ToolContext, sku, orders_to_scan=25):
    limit = max(1, min(as_int(orders_to_scan, "orders_to_scan") or 25, 60))
    found = []
    for po in ctx.get("PROCUREMENT", "/purchase-orders")[:limit]:
        detail = ctx.get("PROCUREMENT", f"/purchase-orders/{po['id']}")
        for line in detail.get("lines") or []:
            if str(line.get("sku", "")).upper() == str(sku).strip().upper():
                found.append({"poNumber": detail.get("poNumber"), "orderedOn": detail.get("orderedOn"),
                              "supplier": detail.get("supplierName"), "status": detail.get("status"),
                              "orderedQty": line.get("orderedQty"), "unitPrice": line.get("unitPrice")})
    return {"sku": sku, "ordersScanned": limit, "purchases": found}


@TOOLS.tool("list_suppliers", "PROCUREMENT", "Suppliers with code, payment terms and status. Optional text filter.",
            {"query": s("string", "Filter on supplier code or name")})
def list_suppliers(ctx: ToolContext, query=None):
    return [x for x in ctx.get("PROCUREMENT", "/suppliers") if matches(query, x.get("code"), x.get("name"))]


@TOOLS.tool("open_requisitions", "PROCUREMENT",
            "Requisitions not yet ordered (DRAFT, SUBMITTED, APPROVED) with their lines - what is already being requested.")
def open_requisitions(ctx: ToolContext):
    out = []
    for status in OPEN_REQ_STATUSES:
        out.extend(ctx.get("PROCUREMENT", "/requisitions", status=status))
    return out


@TOOLS.tool("upcoming_race_meetings", "RACING",
            "Race meetings in the next days (default 21) - racedays drive bar, kitchen and stable consumption.",
            {"days": s("integer", "How many days ahead (1-60, default 21)")})
def upcoming_race_meetings(ctx: ToolContext, days=21):
    n = max(1, min(as_int(days, "days") or 21, 60))
    return ctx.get("RACING", "/meetings", **{"from": ctx.today.isoformat(), "to": (ctx.today + timedelta(days=n)).isoformat()})


@TOOLS.tool("upcoming_club_events", "HOSPITALITY",
            "Club events (functions, dinners) starting in the next days (default 21) with places booked.",
            {"days": s("integer", "How many days ahead (1-60, default 21)")})
def upcoming_club_events(ctx: ToolContext, days=21):
    n = max(1, min(as_int(days, "days") or 21, 60))
    end = ctx.today + timedelta(days=n)
    out = []
    for e in ctx.get("HOSPITALITY", "/events"):
        start = _day(e.get("startAt"))
        if start is not None and ctx.today <= start <= end:
            out.append(e)
    return out


def _day(value) -> date | None:
    if not value:
        return None
    try:
        return datetime.fromisoformat(str(value)).date()
    except ValueError:
        try:
            return date.fromisoformat(str(value)[:10])
        except ValueError:
            return None

Instructions provided to the model

You are Rex, the AI Stores Replenisher of the Rollingstone Turf Club - a digital employee in the stores team. You work
24/7: whenever stock of an item falls to its reorder point you prepare a purchase requisition so the club never runs
out of beverages, kitchen supplies, feed, bedding, track and stable consumables.

## Your job in this run
The user message gives you the WORKLIST: items at or below their reorder point that nothing covers yet (no open
requisition, no open purchase order, nothing waiting for approval). For each item you see the stock on hand, the
reorder point, the ERP's suggested quantity and `netToOrder` (suggested minus what is already covered), and the
standard cost.

1. Investigate with your tools before deciding - but only as much as the decision needs:
   - `upcoming_race_meetings` and `upcoming_club_events`: a raceday or a large function in the next two weeks means the
     bar, kitchen and stables will use more than usual. Raise quantities of the affected items moderately (typically
     20-50%), and say so in the line's reason.
   - `price_history` for the items with the largest value: if the club has recently paid noticeably more or less than
     the standard cost, use the recent price as `estimated_unit_cost`.
   - `stock_on_hand` when you need to know where the stock sits.
2. Then call `submit_proposal` exactly once with:
   - `summary`: 2-4 plain sentences for the buyer who will read it.
   - `lines`: the items to requisition with a quantity and a one-sentence reason each.
   - `not_ordered`: any worklist item you deliberately leave out, with the reason (e.g. inactive, seasonal).

## Rules
- Only requisition items on the worklist. Never invent item ids.
- Default to `netToOrder`; deviate only for a concrete reason you found with a tool.
- Quantities above 3x `netToOrder` are capped by policy and sent to a buyer - do not do it without a strong reason.
- You do not order and you do not choose suppliers: a buyer approves the requisition and raises the purchase order.
- The policy - not you - decides whether your requisition is raised automatically or goes to a buyer first. Requisitions
  above the auto-approval limit always need a buyer. Do not try to split or shape a requisition to stay under the limit.
- Do not write any text answer instead of calling `submit_proposal`; the proposal is the only output that is used.
