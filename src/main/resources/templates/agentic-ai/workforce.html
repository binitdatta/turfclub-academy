Worker entrypoint and task dispatch

"""The stand-alone 24/7 worker: python -m workforce.worker

Runs the scheduler (schedules, watches, escalation, sweeper) and the runs it starts, without the web app. Use it in
production with WORKFORCE_SCHEDULER=off in the web process - or leave both on: a MySQL named lock makes sure only one
of them schedules at a time, and the other takes over if it stops."""
from __future__ import annotations

import logging
import os
import signal
import threading
from pathlib import Path

from dotenv import load_dotenv


def main() -> None:
    load_dotenv(Path(__file__).resolve().parent.parent / ".env")
    logging.basicConfig(level=os.environ.get("LOG_LEVEL", "INFO"), format="%(asctime)s %(levelname)-5s [%(name)s] %(message)s")
    logging.getLogger("apscheduler").setLevel(logging.WARNING)
    from .config import Config
    from .db import Database
    from .employees import load_roster
    from .engine.runner import Engine
    from .engine.scheduler import WorkforceScheduler
    from .platform import Platform

    cfg = Config()
    engine = Engine(cfg, Database(cfg), Platform(cfg), load_roster())
    engine.setup()
    scheduler = WorkforceScheduler(engine)
    scheduler.start()
    stop = threading.Event()

    def on_signal(signum, frame):  # noqa: ARG001
        logging.getLogger(__name__).info("Stopping the workforce worker (signal %s)", signum)
        stop.set()

    signal.signal(signal.SIGINT, on_signal)
    signal.signal(signal.SIGTERM, on_signal)
    for j in scheduler.jobs():
        logging.getLogger(__name__).info("job %-12s next %s  %s", j["id"], j["next"], j["name"])
    while not stop.wait(5):
        pass
    scheduler.shutdown()
    engine.shutdown()


if __name__ == "__main__":
    main()

Shared employee contract and orchestration

"""What an AI employee is: a manifest (who it is, which service account, when it works, who approves its work) plus
four plain-Python hooks the engine calls in order:

    prepare(ctx)                    -> Facts            deterministic worklist from the ERP (no LLM); empty = nothing to do
    [the LLM investigates with the employee's read tools and calls submit_proposal]
    plan(ctx, facts, proposal)      -> [PlannedAction]  deterministic policy: validates the proposal against the facts and
                                                        decides per action AUTO / APPROVAL (by whom) / SKIP
    execute(ectx, action)           -> ActionResult     deterministic executor: the ERP write as the employee's account

The model never writes to the ERP and never decides its own autonomy: it investigates and proposes; code checks the
proposal, a person approves whatever the policy says needs a person, and code carries it out.
"""
from __future__ import annotations

from dataclasses import asdict, dataclass, field
from pathlib import Path
from typing import Any, Callable

from ..platform.tools import Toolbox

AUTO = "AUTO"
APPROVAL = "APPROVAL"
SKIP = "SKIP"


@dataclass
class Facts:
    """The deterministic starting point of a run, given to the model as JSON and kept on the run for the audit."""
    empty: bool
    headline: str
    data: dict[str, Any] = field(default_factory=dict)
    fingerprint: str | None = None          # for watch triggers: only start a run when this changes

    def to_dict(self) -> dict[str, Any]:
        return asdict(self)

    @staticmethod
    def from_dict(d: dict[str, Any] | None) -> "Facts":
        d = d or {}
        return Facts(bool(d.get("empty")), d.get("headline") or "", d.get("data") or {}, d.get("fingerprint"))


@dataclass
class PlannedAction:
    key: str                     # unique within the run: "requisition", "letter-1234", "hold-1234"
    kind: str                    # CREATE_REQUISITION, SEND_LETTER, PLACE_HOLD ...
    title: str
    summary: str
    payload: dict[str, Any]
    mode: str                    # AUTO / APPROVAL / SKIP
    reasons: list[str] = field(default_factory=list)
    amount: float | None = None

    def to_dict(self) -> dict[str, Any]:
        return asdict(self)

    @staticmethod
    def from_dict(d: dict[str, Any]) -> "PlannedAction":
        return PlannedAction(d["key"], d["kind"], d["title"], d.get("summary") or "", d.get("payload") or {}, d["mode"],
                             list(d.get("reasons") or []), d.get("amount"))


@dataclass
class ActionResult:
    status: str                  # DONE / FAILED
    erp_reference: str | None = None
    response: Any = None
    error: str | None = None
    document: dict[str, Any] | None = None    # a letter that was sent: kind, party_id, party_name, recipient, subject, body ...


@dataclass(frozen=True)
class EditField:
    """One field a person may change on the approval page before approving."""
    name: str
    label: str
    kind: str                    # text / textarea / number
    value: Any
    help: str | None = None
    min: float | None = None
    max: float | None = None


class EditError(ValueError):
    """An approver's edit is not acceptable (shown back on the form)."""


@dataclass
class ExecContext:
    api: Any
    cfg: Any
    db: Any
    mailer: Any
    employee: "Employee"
    run_id: int
    approved_by: str | None = None


@dataclass
class Employee:
    code: str
    name: str
    title: str
    description: str
    icon: str
    client_id: str
    reads: tuple[str, ...]                    # ERP modules it reads
    writes: tuple[str, ...]                   # plain-language list of what it may change, and through which endpoint
    write_role: str                           # its one Keycloak write role
    approver_role: str                        # the role that decides its approvals
    approver_label: str
    toolbox: Toolbox
    prompt_file: Path
    proposal_schema: dict[str, Any]           # input_schema of its submit_proposal tool
    prepare: Callable[..., Facts]
    plan: Callable[..., list[PlannedAction]]
    execute: Callable[..., ActionResult]
    edit_fields: Callable[[PlannedAction], list[EditField]]
    apply_edits: Callable[[PlannedAction, dict[str, Any]], dict[str, Any]]
    default_cron: str                         # 5-field crontab in the club's time zone
    watch_minutes: int | None = None          # also poll prepare() every N minutes and start a run when it changes
    escalate_after_minutes: int = 240
    expire_after_hours: int = 72
    minutes_saved_per_action: int = 10
    autonomy: tuple[str, ...] = ()            # the policy in plain language, for the roster and the landing page
    view_roles: tuple[str, ...] = ()          # who may see its runs besides the approver, TC_ADMIN and TC_AUDITOR
    settings: Callable[[Any], list[tuple[str, str]]] = lambda cfg: []

    def secret(self, cfg) -> str:
        return cfg.employee(self.code, "CLIENT_SECRET", f"{self.client_id}-secret")

    def cron(self, cfg) -> str:
        return cfg.employee(self.code, "CRON", self.default_cron)

    def watch_interval(self, cfg) -> int | None:
        value = cfg.employee_int(self.code, "WATCH_MINUTES", self.watch_minutes or 0)
        return value or None

    def escalate_minutes(self, cfg) -> int:
        return cfg.employee_int(self.code, "ESCALATE_AFTER_MINUTES", self.escalate_after_minutes)

    def expire_hours(self, cfg) -> int:
        return cfg.employee_int(self.code, "EXPIRE_AFTER_HOURS", self.expire_after_hours)

    @property
    def prompt(self) -> str:
        return self.prompt_file.read_text(encoding="utf-8")

    def visible_to(self, user) -> bool:
        return user is not None and (user.is_auditor_or_admin or user.has_any(self.approver_role, *self.view_roles))

    def can_approve(self, user) -> bool:
        return user is not None and user.has_any(self.approver_role)

    def can_operate(self, user) -> bool:
        """Run now: the approver of its work or TC_ADMIN."""
        return user is not None and (user.is_admin or user.has_any(self.approver_role))

def _env(name: str, default: str | None = None) -> str | None:
    value = os.environ.get(name)
    return value if value not in (None, "") else default


def _int(name: str, default: int) -> int:
    return int(_env(name, str(default)))

User access constraints for workforce UI

"""Who may see and decide what. The rules, in one place:

* An approval is decided by a person holding its required role (e.g. PROCUREMENT_MANAGE for the replenisher, AR_MANAGE
  for the collections officer). TC_ADMIN holds every MANAGE role (Keycloak composite), so escalations reach it.
  TC_AUDITOR sees everything and decides nothing.
* An employee's runs, actions and letters are visible to its approvers, its view roles, TC_ADMIN and TC_AUDITOR.
* Switching an employee on or off is TC_ADMIN only; "Run now" is its approvers and TC_ADMIN.
* The usage and cost dashboard is TC_ADMIN / TC_AUDITOR, as in TurfBot.
"""
from __future__ import annotations

from flask import abort, current_app

from .auth.session import User


def engine():
    return current_app.extensions["workforce.engine"]


def store():
    return engine().store


def db():
    return current_app.extensions["workforce.db"]


def roster():
    return engine().roster


def cfg():
    return current_app.config["WORKFORCE"]


def visible_codes(user: User | None) -> list[str]:
    return [e.code for e in roster().values() if e.visible_to(user)]


def require_visible(user: User | None, code: str) -> None:
    if code not in roster() or code not in visible_codes(user):
        abort(404 if code not in roster() else 403)


def decidable_roles(user: User | None) -> list[str]:
    if user is None:
        return []
    return sorted({e.approver_role for e in roster().values() if user.has_any(e.approver_role)})


def pending_for(user: User | None) -> int:
    roles = decidable_roles(user)
    if not roles:
        return 0
    marks = ", ".join(["%s"] * len(roles))
    try:
        return int(db().scalar(f"SELECT COUNT(*) AS n FROM approval WHERE status = 'PENDING' AND required_role IN ({marks})",
                               roles) or 0)
    except Exception:  # noqa: BLE001 - the badge must never break a page
        return 0
